package main

import (
	"errors"
	"fmt"
)

type Turma struct {
	Codigo string
	Vagas  int
	Alunos []string
}

func (t *Turma) Matricular(nome string) error {
	if t.Vagas == 0 {
		return errors.New("turma cheia")
	}
	t.Vagas--
	t.Alunos = append(t.Alunos, nome)
	return nil
}

func main() {
	turmas := map[string]Turma{"GO-01": {Codigo: "GO-01", Vagas: 2}}
	turmas["GO-01"].Matricular("Ana")
	fmt.Println(turmas)
}
